**Technical Implementation Details:**
```csharp
// Added to PhotoService.cs
private readonly IHttpContextAccessor _httpContextAccessor;

// Inject in constructor
public PhotoService(IHttpContextAccessor httpContextAccessor, ...)
{
    _httpContextAccessor = httpContextAccessor;
}

// New method for complete URL generation
private PhotoUrlsDto GeneratePhotoUrls(int photoId)
{
    var request = _httpContextAccessor.HttpContext?.Request;
    var baseUrl = $"{request?.Scheme}://{request?.Host}";
    
    return new PhotoUrlsDto
    {
        OriginalUrl = $"{baseUrl}/api/photos/{photoId}/image",
        ThumbnailUrl = $"{baseUrl}/api/photos/{photoId}/thumbnail"
    };
}
```
